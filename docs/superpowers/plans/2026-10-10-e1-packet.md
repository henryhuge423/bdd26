## Review Focus

1. **窗口裁剪越界**：质心近边/角时 160×160 窗口必须 clamp 并保持尺寸不变。→ Task 2 测试四角。

def test_stratified_sample_counts_probs_merge():
    rows = ([{"tissue": 0, "border": True, "k": i} for i in range(100)] +
            [{"tissue": 0, "border": False, "k": i} for i in range(100, 300)] +
            [{"tissue": 1, "border": False, "k": i} for i in range(300, 320)])

```python
"""E1 盲审抽样与绘图纯逻辑。spec: docs/superpowers/specs/2026-10-10-e1-blind-audit-design.md。"""
from __future__ import annotations

import cv2
import numpy as np

WINDOW = 160
MERGE_MIN = 3


def _alloc(quota_by_cell: dict, n_total: int) -> dict:
    """Largest-remainder integer allocation keeping cells with quota < MERGE_MIN merged upstream."""
    raw = {k: n_total * N / sum(quota_by_cell.values()) for k, N in quota_by_cell.items()}
    base = {k: int(v) for k, v in raw.items()}
    rem = n_total - sum(base.values())
    for k in sorted(raw, key=lambda k: raw[k] - base[k], reverse=True)[:rem]:
        base[k] += 1
    return base


def stratified_sample(rows, n_total, rng, merge_min=MERGE_MIN):
    cells = {}
    for r in rows:
        cells.setdefault((r["tissue"], bool(r["border"])), []).append(r)
    # pre-registered merge: a cell whose quota < merge_min joins the same tissue's other border cell
    for key in list(cells):
        if key in cells:
            pass
    sizes = {k: len(v) for k, v in cells.items()}
    pre = {k: n_total * N / sum(sizes.values()) for k, N in sizes.items()}
    for key in [k for k, q in pre.items() if q < merge_min]:
        partner = (key[0], not key[1])
        if partner in cells:
            cells[partner].extend(cells.pop(key))
    sizes = {k: len(v) for k, v in cells.items()}
    alloc = _alloc(sizes, n_total)
    out = []
    for key, members in sorted(cells.items()):
        n = min(alloc[key], len(members))
        idx = rng.choice(len(members), n, replace=False)
        for i in idx:
            r = dict(members[i]); r["prob"] = n / len(members)
            out.append(r)
    return out


def background_windows(tissue_arr, n_total, rng, shape=(256, 256), size=WINDOW):
    """S4: per-tissue allocate n_t, sample images WITHOUT replacement, one uniform center each.
    pi_i = (n_t / N_t) * (1 / C), C = (#valid centers) = (shape - size + 1)^2 for size 160 on 256 -> 96*96."""
    by_t = {t: np.nonzero(tissue_arr == t)[0] for t in np.unique(tissue_arr)}
    alloc = _alloc({t: len(v) for t, v in by_t.items()}, n_total)
    C = (shape[0] - size + 1) * (shape[1] - size + 1)
    out = []
    for t, imgs in sorted(by_t.items()):
        n = min(alloc[t], len(imgs))
        for img in rng.choice(imgs, n, replace=False):
            cy = int(rng.integers(size // 2, shape[0] - size // 2 + 1))
            cx = int(rng.integers(size // 2, shape[1] - size // 2 + 1))
            out.append({"image": int(img), "cy": cy, "cx": cx, "tissue": int(t),
                        "prob": (n / len(imgs)) / C})
    return out
```

（S4 的分析单位是窗口，π 按"选图×均匀中心"两步设计计算；测试断言每窗 `prob == n_t/N_t/C`。）
`window_slice` / `draw_cross` / `overlay_outlines` 按 Interfaces 区签名用 cv2 实现（各 ≤8 行：
`window_slice` 返回恒定 160×160 的两个 slice 与窗口左上偏移；`draw_cross` 画 9px 红十字；
`overlay_outlines` 对每个掩码 `cv2.findContours` + `cv2.polylines` 上指定颜色）。
